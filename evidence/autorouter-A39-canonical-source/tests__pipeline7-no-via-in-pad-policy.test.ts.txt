import { expect, test } from "bun:test"
import { createPipeline7AutoroutingDrcEvaluator } from "lib/autorouter-pipelines/AutoroutingPipeline7_MultiGraph/create-pipeline7-autorouting-drc-evaluator"
import type { SimpleRouteJson } from "lib/types"
import type { HighDensityRoute } from "lib/types/high-density-types"
import { getConnectivityMapFromSimpleRouteJson } from "lib/utils/getConnectivityMapFromSimpleRouteJson"

test("Pipeline7 explicit no-via-in-pad policy enforces connected-pad clearance", () => {
  const srj: SimpleRouteJson = {
    layerCount: 4,
    minTraceWidth: 0.2,
    minViaEdgeToPadEdgeClearance: 0.2,
    bounds: { minX: -2, minY: -2, maxX: 2, maxY: 2 },
    obstacles: [{
      type: "rect",
      layers: ["top"],
      center: { x: 0, y: 0 },
      width: 0.4,
      height: 0.4,
      connectedTo: ["pcb_smtpad_connected", "connected"],
    }],
    connections: [{
      name: "connected",
      pointsToConnect: [
        { x: 0, y: 0, layer: "top", pointId: "pad" },
        { x: 1, y: 0, layer: "bottom", pointId: "other_end" },
      ],
    }],
  }
  const routes: HighDensityRoute[] = [{
    connectionName: "connected",
    route: [
      { x: 0, y: 0, z: 0 },
      { x: 0.5, y: 0, z: 0 },
      { x: 0.5, y: 0, z: 3 },
      { x: 1, y: 0, z: 3 },
    ],
    vias: [{ x: 0.5, y: 0 }],
    traceThickness: 0.2,
    viaDiameter: 0.6,
  }]
  for (const allowViaInPad of [undefined, true, false]) {
    const configuredSrj = { ...srj, allowViaInPad }
    const evaluator = createPipeline7AutoroutingDrcEvaluator({
      connections: srj.connections,
      originalConnections: srj.connections,
      layerCount: 4,
      obstacles: srj.obstacles,
      defaultViaHoleDiameter: 0.3,
      connMap: getConnectivityMapFromSimpleRouteJson(configuredSrj),
      srjWithPointPairs: configuredSrj,
      originalSrj: configuredSrj,
    })
    const result = evaluator({ traces: [], routes })
    if (Array.isArray(result)) throw new Error("Expected centered DRC result")
    expect(result.errors).toHaveLength(allowViaInPad === false ? 1 : 0)
    if (allowViaInPad === false) {
      expect(result.errors[0]).toMatchObject({
        type: "pcb_pad_pad_clearance_error",
        minimum_clearance: 0.2,
        actual_clearance: 0,
      })
    }
  }
})
